SELECT LAST_NAME||' '||SALARY AS wynagrodzenie  
FROM EMPLOYEES 
WHERE 1=1
		AND DEPARTMENT_ID IN (20,50)
		AND SALARY BETWEEN 2000 AND 7000
		ORDER BY LAST_NAME 

SELECT HIRE_DATE ,LAST_NAME, ${col}
FROM EMPLOYEES
WHERE 1=1
	AND MANAGER_ID IS NOT NULL 
	AND EXTRACT(YEAR FROM HIRE_DATE) = 2005
	ORDER BY ${col}
