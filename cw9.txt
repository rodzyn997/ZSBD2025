################## Zad 1 ##################
CREATE OR REPLACE PACKAGE pkg_zajecia IS
    FUNCTION pesel_to_date(p_pesel VARCHAR2) RETURN DATE;
    
    PROCEDURE add_job(p_job_id jobs.job_id%TYPE, p_job_title jobs.job_title%TYPE);
    
    FUNCTION get_job_name(p_job_id jobs.job_id%TYPE) RETURN VARCHAR2;
END pkg_zajecia;

CREATE OR REPLACE PACKAGE BODY pkg_zajecia IS

    FUNCTION pesel_to_date(p_pesel VARCHAR2) RETURN DATE IS
        v_year  NUMBER := SUBSTR(p_pesel, 1, 2);
        v_month NUMBER := SUBSTR(p_pesel, 3, 2);
        v_day   NUMBER := SUBSTR(p_pesel, 5, 2);
    BEGIN
        IF v_month > 80 THEN v_year := 1800 + v_year; v_month := v_month - 80;
        ELSIF v_month > 20 THEN v_year := 2000 + v_year; v_month := v_month - 20;
        ELSE v_year := 1900 + v_year; END IF;
        RETURN TO_DATE(v_year || '-' || v_month || '-' || v_day, 'YYYY-MM-DD');
    END;

    PROCEDURE add_job(p_job_id jobs.job_id%TYPE, p_job_title jobs.job_title%TYPE) IS
    BEGIN
        INSERT INTO jobs (job_id, job_title) VALUES (p_job_id, p_job_title);
        COMMIT;
    EXCEPTION WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('Błąd w add_job: ' || SQLERRM);
    END;

    FUNCTION get_job_name(p_job_id jobs.job_id%TYPE) RETURN VARCHAR2 IS
        v_title jobs.job_title%TYPE;
    BEGIN
        SELECT job_title INTO v_title FROM jobs WHERE job_id = p_job_id;
        RETURN v_title;
    EXCEPTION WHEN NO_DATA_FOUND THEN RETURN 'Nieznany';
    END;

END pkg_zajecia;

CREATE OR REPLACE PACKAGE pkg_regions IS
    e_region_exists EXCEPTION;
    PRAGMA EXCEPTION_INIT(e_region_exists, -20001);
    
    e_region_has_countries EXCEPTION;
    PRAGMA EXCEPTION_INIT(e_region_has_countries, -20002);

    -- CRUD Procedury
    PROCEDURE create_region(p_id NUMBER, p_name VARCHAR2);
    PROCEDURE read_region(p_id NUMBER); 
    PROCEDURE read_region(p_name VARCHAR2); 
    PROCEDURE update_region(p_id NUMBER, p_new_name VARCHAR2);
    PROCEDURE delete_region(p_id NUMBER);
END pkg_regions;

################## Zad 3 ##################

CREATE OR REPLACE PACKAGE BODY pkg_regions IS

    PROCEDURE log_error(p_code VARCHAR2, p_msg VARCHAR2, p_proc VARCHAR2) IS
        PRAGMA AUTONOMOUS_TRANSACTION; 
    BEGIN
        INSERT INTO logi_bledow (kod_bledu, komunikat, procedura)
        VALUES (p_code, p_msg, p_proc);
        COMMIT;
    END;

    -- CREATE
    PROCEDURE create_region(p_id NUMBER, p_name VARCHAR2) IS
        v_check NUMBER;
    BEGIN
        -- Sprawdzenie czy nazwa istnieje
        SELECT COUNT(*) INTO v_check FROM regions WHERE UPPER(region_name) = UPPER(p_name);
        IF v_check > 0 THEN
            RAISE e_region_exists;
        END IF;

        INSERT INTO regions (region_id, region_name) VALUES (p_id, p_name);
        COMMIT;
        DBMS_OUTPUT.PUT_LINE('Dodano region: ' || p_name);
    EXCEPTION
        WHEN e_region_exists THEN
            log_error('-20001', 'Próba dodania duplikatu: ' || p_name, 'create_region');
            RAISE_APPLICATION_ERROR(-20001, 'Błąd: Region o nazwie ' || p_name || ' już istnieje!');
        WHEN OTHERS THEN
            log_error(SQLCODE, SQLERRM, 'create_region');
            RAISE;
    END;

    -- READ (po ID)
    PROCEDURE read_region(p_id NUMBER) IS
        r_region regions%ROWTYPE;
    BEGIN
        SELECT * INTO r_region FROM regions WHERE region_id = p_id;
        DBMS_OUTPUT.PUT_LINE('ID: ' || r_region.region_id || ', Nazwa: ' || r_region.region_name);
    EXCEPTION WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('Nie znaleziono regionu ID: ' || p_id);
    END;

    -- READ (po Nazwie)
    PROCEDURE read_region(p_name VARCHAR2) IS
    BEGIN
        FOR r IN (SELECT * FROM regions WHERE UPPER(region_name) LIKE UPPER('%' || p_name || '%')) LOOP
            DBMS_OUTPUT.PUT_LINE('Znaleziono: ID ' || r.region_id || ' - ' || r.region_name);
        END LOOP;
    END;

    -- UPDATE
    PROCEDURE update_region(p_id NUMBER, p_new_name VARCHAR2) IS
    BEGIN
        UPDATE regions SET region_name = p_new_name WHERE region_id = p_id;
        IF SQL%ROWCOUNT = 0 THEN
            DBMS_OUTPUT.PUT_LINE('Brak regionu do aktualizacji.');
        ELSE
            COMMIT;
            DBMS_OUTPUT.PUT_LINE('Zaktualizowano region ID ' || p_id);
        END IF;
    END;

    -- DELETE
    PROCEDURE delete_region(p_id NUMBER) IS
        v_count NUMBER;
    BEGIN
        -- Sprawdzenie czy ma kraje
        SELECT COUNT(*) INTO v_count FROM countries WHERE region_id = p_id;
        IF v_count > 0 THEN
            RAISE e_region_has_countries;
        END IF;

        DELETE FROM regions WHERE region_id = p_id;
        COMMIT;
        DBMS_OUTPUT.PUT_LINE('Usunięto region ID ' || p_id);
    EXCEPTION
        WHEN e_region_has_countries THEN
            log_error('-20002', 'Próba usunięcia regionu z krajami ID: ' || p_id, 'delete_region');
            RAISE_APPLICATION_ERROR(-20002, 'Nie można usunąć: Region ma przypisane kraje!');
        WHEN OTHERS THEN
            log_error(SQLCODE, SQLERRM, 'delete_region');
            RAISE;
    END;

END pkg_regions;

CREATE OR REPLACE PACKAGE pkg_statystyki IS
    FUNCTION srednia_w_dziale(p_dept_id NUMBER) RETURN NUMBER;
    
    FUNCTION min_max_stanowisko(p_job_id VARCHAR2) RETURN VARCHAR2;
    

    PROCEDURE generuj_raport_departamentu(p_dept_id NUMBER);
END pkg_statystyki;


CREATE OR REPLACE PACKAGE BODY pkg_statystyki IS

    FUNCTION srednia_w_dziale(p_dept_id NUMBER) RETURN NUMBER IS
        v_avg NUMBER;
    BEGIN
        SELECT AVG(salary) INTO v_avg FROM employees WHERE department_id = p_dept_id;
        RETURN NVL(v_avg, 0); 
    END;

    FUNCTION min_max_stanowisko(p_job_id VARCHAR2) RETURN VARCHAR2 IS
        v_min NUMBER;
        v_max NUMBER;
    BEGIN
        SELECT MIN(salary), MAX(salary) INTO v_min, v_max 
        FROM employees WHERE job_id = p_job_id;
        RETURN 'Min: ' || v_min || ', Max: ' || v_max;
    END;

    PROCEDURE generuj_raport_departamentu(p_dept_id NUMBER) IS
        v_dept_name departments.department_name%TYPE;
        v_count NUMBER;
    BEGIN
        SELECT department_name INTO v_dept_name FROM departments WHERE department_id = p_dept_id;
        SELECT COUNT(*) INTO v_count FROM employees WHERE department_id = p_dept_id;
        
        DBMS_OUTPUT.PUT_LINE('=== RAPORT DLA: ' || v_dept_name || ' ===');
        DBMS_OUTPUT.PUT_LINE('Liczba pracowników: ' || v_count);
        DBMS_OUTPUT.PUT_LINE('Średnia pensja:     ' || ROUND(srednia_w_dziale(p_dept_id), 2));
        DBMS_OUTPUT.PUT_LINE('===================================');
    EXCEPTION WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('Departament o ID ' || p_dept_id || ' nie istnieje.');
    END;

END pkg_statystyki;

CREATE OR REPLACE PACKAGE pkg_utils IS
    PROCEDURE napraw_telefony;
    
    PROCEDURE masowa_podwyzka(p_job_id VARCHAR2, p_procent NUMBER);
END pkg_utils;

CREATE OR REPLACE PACKAGE BODY pkg_utils IS

    PROCEDURE napraw_telefony IS
    BEGIN
        UPDATE employees
        SET phone_number = REPLACE(phone_number, '.', '-')
        WHERE phone_number LIKE '%.%';
        
        DBMS_OUTPUT.PUT_LINE('Zaktualizowano ' || SQL%ROWCOUNT || ' numerów telefonów.');
        COMMIT;
    END;

    PROCEDURE masowa_podwyzka(p_job_id VARCHAR2, p_procent NUMBER) IS
    BEGIN
        IF p_procent <= 0 THEN
            DBMS_OUTPUT.PUT_LINE('Procent musi być dodatni!');
            RETURN;
        END IF;

        UPDATE employees
        SET salary = salary * (1 + p_procent/100)
        WHERE job_id = p_job_id;
        
        DBMS_OUTPUT.PUT_LINE('Przyznano podwyżkę dla ' || SQL%ROWCOUNT || ' pracowników na stanowisku ' || p_job_id);
        COMMIT;
    END;

END pkg_utils;
